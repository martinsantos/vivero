 * Enqueue custom redesign CSS
 * Added: October 2025
 */
function loscocos_enqueue_custom_redesign() {
    wp_enqueue_style(
        'loscocos-custom-redesign',
        get_template_directory_uri() . '/assets/css/custom-redesign.css',
        array('loscocos-style'),
        '1.0.0',
        'all'
    );
}
